#define _WINSOCK_DEPRECATED_NO_WARNINGS
#include <winsock2.h>
#include <ws2tcpip.h>
#include <iostream>
#include <string>
#include <windows.h>

#pragma comment(lib, "ws2_32.lib")
using namespace std;

struct Person {
    char name[200];
    int age;
};

int main(int argc, char* argv[]) {
    SetConsoleCP(1251);
    SetConsoleOutputCP(1251);
    setlocale(LC_ALL, "rus");

    string serverIP;

    if (argc >= 2) {
        serverIP = argv[1];
    }
    else {
        cout << "Введите IP адрес сервера (Enter для 127.0.0.1): ";
        getline(cin, serverIP);
        if (serverIP.empty()) {
            serverIP = "127.0.0.1";
        }
    }

    WSADATA wsaData;
    if (WSAStartup(MAKEWORD(2, 2), &wsaData) != 0) {
        cerr << "WSAStartup failed" << endl;
        return 1;
    }

    // Создаём TCP сокет
    SOCKET clientSocket = socket(AF_INET, SOCK_STREAM, 0);
    if (clientSocket == INVALID_SOCKET) {
        cerr << "Socket creation failed" << endl;
        WSACleanup();
        return 1;
    }

    sockaddr_in serverAddr{};
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(12345);

    if (inet_pton(AF_INET, serverIP.c_str(), &serverAddr.sin_addr) <= 0) {
        cerr << "Invalid IP address: " << serverIP << endl;
        closesocket(clientSocket);
        WSACleanup();
        return 1;
    }

    // Подключаемся к серверу
    if (connect(clientSocket, (sockaddr*)&serverAddr, sizeof(serverAddr)) == SOCKET_ERROR) {
        cerr << "Connection failed to " << serverIP << ":12345" << endl;
        cerr << "Error code: " << WSAGetLastError() << endl;
        closesocket(clientSocket);
        WSACleanup();
        return 1;
    }

    cout << "========================================" << endl;
    cout << "TCP ЭХО-КЛИЕНТ" << endl;
    cout << "Подключен к серверу " << serverIP << ":12345" << endl;
    cout << "========================================" << endl;
    cout << "Команды:" << endl;
    cout << "  1 - отправить текст" << endl;
    cout << "  2 - отправить структуру (ФИО + возраст)" << endl;
    cout << "  0 - выйти" << endl;
    cout << "========================================" << endl;

    char buffer[1024];
    int choice;
    int bytesReceived;

    while (true) {
        cout << "\nВыберите действие: ";
        cin >> choice;
        cin.ignore();

        if (choice == 0) break;

        else if (choice == 1) {
            string msg;
            cout << "Введите сообщение: ";
            getline(cin, msg);

            // Отправляем текст
            send(clientSocket, msg.c_str(), msg.length() + 1, 0);

            // Получаем эхо
            bytesReceived = recv(clientSocket, buffer, sizeof(buffer) - 1, 0);
            if (bytesReceived > 0) {
                buffer[bytesReceived] = '\0';
                cout << "Эхо от сервера: " << buffer << endl;
            }
        }

        else if (choice == 2) {
            Person p;
            cout << "Введите ФИО: ";
            cin.getline(p.name, 200);
            cout << "Введите возраст: ";
            cin >> p.age;
            cin.ignore();

            // Отправляем структуру
            send(clientSocket, (char*)&p, sizeof(Person), 0);

            // Получаем эхо (структуру обратно)
            bytesReceived = recv(clientSocket, buffer, sizeof(buffer) - 1, 0);
            if (bytesReceived == sizeof(Person)) {
                Person* ep = (Person*)buffer;
                cout << "Эхо от сервера (структура):" << endl;
                cout << "  ФИО: " << ep->name << endl;
                cout << "  Возраст: " << ep->age << endl;
            }
        }

        else {
            cout << "Неверный выбор!" << endl;
        }
    }

    closesocket(clientSocket);
    WSACleanup();
    cout << "Клиент завершил работу." << endl;
    return 0;
}

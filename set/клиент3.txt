// client.cpp
#define _WINSOCK_DEPRECATED_NO_WARNINGS

#include <winsock2.h>
#include <iostream>
#include <string>

#pragma comment(lib, "ws2_32.lib")

using namespace std;

int main() {
    setlocale(LC_ALL, "ru");

    cout << "=== C++ HTTP КЛИЕНТ ===" << endl;
    cout << "1 - json.org (веб-сервер)" << endl;
    cout << "2 - localhost:5555 (свой сервер)" << endl;
    cout << "Выбор: ";

    int choice;
    cin >> choice;

    WSADATA wsaData;
    WSAStartup(MAKEWORD(2, 2), &wsaData);

    SOCKET sock = socket(AF_INET, SOCK_STREAM, 0);

    sockaddr_in server;
    server.sin_family = AF_INET;

    if (choice == 1) {
        server.sin_port = htons(80);
        struct hostent* host = gethostbyname("json.org");
        if (host == NULL) {
            cout << "Ошибка: не найден json.org" << endl;
            return 1;
        }
        memcpy(&server.sin_addr, host->h_addr_list[0], host->h_length);
        cout << "\nПодключение к json.org:80..." << endl;
    }
    else {
        server.sin_port = htons(5555);
        server.sin_addr.s_addr = inet_addr("127.0.0.1");
        cout << "\nПодключение к localhost:5555..." << endl;
    }

    if (connect(sock, (SOCKADDR*)&server, sizeof(server)) == SOCKET_ERROR) {
        cout << "Ошибка подключения!" << endl;
        return 1;
    }
    cout << "Подключено!" << endl;

    // Запрос
    string request =
        "GET / HTTP/1.1\r\n"
        "Host: " + string(choice == 1 ? "json.org" : "localhost") + "\r\n"
        "Connection: close\r\n"
        "\r\n";

    cout << "\n--- Отправка запроса ---" << endl;
    cout << request;
    cout << "------------------------" << endl;

    send(sock, request.c_str(), request.size(), 0);

    // Получение и вывод ответа
    char buffer[8192];
    int bytes;
    cout << "\n--- Ответ сервера ---" << endl;

    while ((bytes = recv(sock, buffer, sizeof(buffer) - 1, 0)) > 0) {
        buffer[bytes] = '\0';
        cout << buffer;
    }

    cout << "\n--- Конец ответа ---" << endl;

    closesocket(sock);
    WSACleanup();

    cout << "\nНажмите Enter для выхода...";
    cin.ignore();
    cin.get();
    return 0;
}
